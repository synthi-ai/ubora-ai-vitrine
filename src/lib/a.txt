const FeatureCard = ({ feature, index }) => {
  const cardRef = useRef(null);
  const imageRef = useRef(null);

  useEffect(() => {
    gsap.registerPlugin(ScrollTrigger);

    // Animation de la carte
    gsap.from(cardRef.current, {
      scrollTrigger: {
        trigger: cardRef.current,
        start: "top center+=100",
        toggleActions: "play none none reverse",
      },
      opacity: 0,
      y: 50,
      rotationX: -30,
      duration: 1,
      ease: "power3.out",
    });

    // Animation de l'image
    gsap.to(imageRef.current, {
      scrollTrigger: {
        trigger: imageRef.current,
        start: "top center",
        scrub: 1,
      },
      rotationY: 360,
      transformPerspective: 1000,
      ease: "none",
      repeat: -1,
    });

    // Animation au survol
    const hoverAnimation = gsap.to(cardRef.current, {
      scale: 1.05,
      rotationY: 5,
      boxShadow: "0 20px 40px rgba(0,0,0,0.2)",
      duration: 0.3,
      paused: true,
    });

    cardRef.current.addEventListener("mouseenter", () => hoverAnimation.play());
    cardRef.current.addEventListener("mouseleave", () => hoverAnimation.reverse());

    return () => {
      ScrollTrigger.getAll().forEach(t => t.kill());
    };
  }, []);

  return (
    <div
      ref={cardRef}
      className="bg-white rounded-2xl p-8 shadow-lg transform-gpu"
      style={{ perspective: "1000px" }}
    >
      <div className="flex items-start gap-6">
        <div
          ref={imageRef}
          className="w-24 h-24 flex-shrink-0 relative"
        >
          <img
            src={feature.image}
            alt={feature.title}
            className="w-full h-full object-contain"
          />
        </div>
        <div>
          <h3 className="text-xl font-bold mb-3 text-gray-900">{feature.title}</h3>
          <p className="text-gray-600 leading-relaxed">{feature.description}</p>
        </div>
      </div>
    </div>
  );
};

const Visualization = () => {
  const containerRef = useRef(null);

  useEffect(() => {
    const elements = containerRef.current.children;
    
    gsap.set(elements, {
      transformPerspective: 1000,
    });

    gsap.to(elements, {
      rotationY: 360,
      duration: 20,
      stagger: 2,
      repeat: -1,
      ease: "none",
    });

    // Animation des particules
    const particles = [...Array(10)].map((_, i) => {
      const particle = document.createElement('div');
      particle.className = 'absolute w-2 h-2 bg-indigo-500 rounded-full';
      containerRef.current.appendChild(particle);

      gsap.set(particle, {
        x: Math.random() * 400 - 200,
        y: Math.random() * 400 - 200,
        opacity: 0,
      });

      gsap.to(particle, {
        duration: 2 + Math.random() * 2,
        opacity: 0.5,
        scale: Math.random() * 2,
        repeat: -1,
        yoyo: true,
        ease: "power1.inOut",
      });

      return particle;
    });

    return () => {
      particles.forEach(p => p.remove());
    };
  }, []);

  return (
    <div
      ref={containerRef}
      className="relative w-full h-96 bg-gradient-to-r from-indigo-50 to-purple-50 rounded-xl overflow-hidden"
    >
      <div className="absolute inset-0 flex items-center justify-center">
        <div className="text-center">
          <h4 className="text-2xl font-bold text-indigo-900 mb-4">
            Visualisation 3D
          </h4>
          <p className="text-indigo-600">
            Explorez nos fonctionnalités en profondeur
          </p>
        </div>
      </div>
    </div>
  );
};
