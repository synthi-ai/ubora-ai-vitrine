interface UseCaseCardProps {
  useCase: UseCase;
  index: number;
  isActive: boolean;
  onHover: () => void;
}

const UseCaseCard = ({ useCase, index, isActive, onHover }: UseCaseCardProps) => {
  const cardRef = useRef<HTMLDivElement>(null);
  const [showDetails, setShowDetails] = useState(false);

  useEffect(() => {
    if (!cardRef.current) return;

    // Entrance animation
    gsap.fromTo(cardRef.current,
      {
        opacity: 0,
        y: 80,
        rotationX: -20,
        scale: 0.9
      },
      {
        scrollTrigger: {
          trigger: cardRef.current,
          start: "top 85%",
          toggleActions: "play none none reverse"
        },
        opacity: 1,
        y: 0,
        rotationX: 0,
        scale: 1,
        duration: 0.8,
        delay: index * 0.15,
        ease: "power3.out"
      }
    );
  }, [index]);

  return (
    <motion.div
      ref={cardRef}
      className={`relative group cursor-pointer transform-gpu ${
        isActive ? 'lg:scale-105 lg:z-10' : ''
      }`}
      onMouseEnter={() => {
        onHover();
        setShowDetails(true);
      }}
      onMouseLeave={() => setShowDetails(false)}
      whileHover={{ y: -8 }}
      layout
    >
      <div className={`relative bg-white rounded-2xl overflow-hidden shadow-xl border-2 transition-all duration-500 ${
        isActive 
          ? 'border-[#8cd139] shadow-2xl' 
          : 'border-white/20 hover:border-gray-200 hover:shadow-2xl'
      }`}>
        {/* Gradient Background */}
        <div className={`absolute inset-0 bg-gradient-to-br ${useCase.gradient} opacity-5 group-hover:opacity-10 transition-opacity duration-500`} />

        {/* Industry Badge */}
        <div className="absolute top-4 right-4 z-10">
          <span className={`inline-block px-3 py-1 rounded-full text-xs font-semibold text-white bg-gradient-to-r ${useCase.gradient}`}>
            {useCase.industry}
          </span>
        </div>

        <div className="relative p-8 z-10">
          {/* Icon Section */}
          <div className="mb-6">
            <motion.div 
              className={`w-16 h-16 rounded-2xl bg-gradient-to-br ${useCase.gradient} p-4 flex items-center justify-center text-white shadow-lg`}
              whileHover={{ scale: 1.1, rotate: 5 }}
              transition={{ type: "spring", stiffness: 300, damping: 20 }}
            >
              <useCase.icon size={32} />
            </motion.div>
          </div>

          {/* Content */}
          <div>
            <h3 className="text-xl lg:text-2xl font-bold mb-4 text-gray-900 leading-tight">
              {useCase.title}
            </h3>
            
            <p className="text-gray-600 mb-6 leading-relaxed">
              {useCase.description}
            </p>

            {/* Stats Grid */}
            <div className="grid grid-cols-2 gap-3 mb-6">
              {Object.entries(useCase.stats).map(([key, value]) => (
                <div key={key} className="text-center p-3 bg-gray-50 rounded-xl border border-gray-100">
                  <div className={`text-lg font-bold bg-gradient-to-r ${useCase.gradient} bg-clip-text text-transparent`}>
                    {value}
                  </div>
                  <div className="text-xs text-gray-500 capitalize font-medium">
                    {key.replace(/([A-Z])/g, ' $1').trim()}
                  </div>
                </div>
              ))}
            </div>

            {/* Agentic Capabilities Preview */}
            <div className="mb-6">
              <h4 className="flex items-center gap-2 font-semibold text-gray-900 mb-3">
                <Bot size={16} className="text-[#8cd139]" />
                AI Agent Capabilities
              </h4>
              <div className="space-y-2">
                {useCase.agenticCapabilities.slice(0, 2).map((capability, idx) => (
                  <div key={idx} className="flex items-start gap-2">
                    <Check size={16} className="text-[#8cd139] mt-0.5 flex-shrink-0" />
                    <span className="text-sm text-gray-700 font-medium">{capability.name}</span>
                  </div>
                ))}
                <div className="text-sm text-gray-500">
                  +{useCase.agenticCapabilities.length - 2} more capabilities
                </div>
              </div>
            </div>

            {/* CTA */}
            <motion.button
              className={`group w-full flex items-center justify-center gap-2 py-3 px-4 rounded-xl font-semibold transition-all duration-300 ${
                isActive
                  ? `bg-gradient-to-r ${useCase.gradient} text-white shadow-lg`
                  : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
              }`}
              whileHover={{ scale: 1.02 }}
              whileTap={{ scale: 0.98 }}
              onClick={() => setShowDetails(!showDetails)}
            >
              <span>Explore Solution</span>
              <ChevronRight size={16} className="group-hover:translate-x-1 transition-transform" />
            </motion.button>
          </div>
        </div>

        {/* Detailed View Modal/Overlay */}
        <AnimatePresence>
          {showDetails && (
            <motion.div
              className="absolute inset-0 bg-white rounded-2xl z-20 overflow-hidden"
              initial={{ opacity: 0, scale: 0.95 }}
              animate={{ opacity: 1, scale: 1 }}
              exit={{ opacity: 0, scale: 0.95 }}
              transition={{ duration: 0.3 }}
            >
              <div className="p-6 h-full overflow-y-auto">
                <div className="flex items-center justify-between mb-4">
                  <h3 className="text-xl font-bold text-gray-900">{useCase.title}</h3>
                  <button
                    onClick={() => setShowDetails(false)}
                    className="w-8 h-8 rounded-full bg-gray-100 flex items-center justify-center hover:bg-gray-200"
                  >
                    ×
                  </button>
                </div>

                {/* Agentic Capabilities */}
                <div className="mb-6">
                  <h4 className="font-semibold text-gray-900 mb-3">Agentic AI Capabilities</h4>
                  <div className="space-y-3">
                    {useCase.agenticCapabilities.map((capability, idx) => (
                      <div key={idx} className="flex items-start gap-3 p-3 bg-gray-50 rounded-lg">
                        <capability.icon size={20} className="text-[#8cd139] mt-0.5" />
                        <div>
                          <div className="font-medium text-gray-900">{capability.name}</div>
                          <div className="text-sm text-gray-600">{capability.description}</div>
                        </div>
                      </div>
                    ))}
                  </div>
                </div>

                {/* Automation Workflows */}
                <div className="mb-6">
                  <h4 className="font-semibold text-gray-900 mb-3">Automated Workflows</h4>
                  <div className="space-y-2">
                    {useCase.automationWorkflows.map((workflow, idx) => (
                      <div key={idx} className="flex items-center gap-2">
                        <Zap size={14} className="text-[#8cd139]" />
                        <span className="text-sm text-gray-700">{workflow}</span>
                      </div>
                    ))}
                  </div>
                </div>

                {/* Real World Example */}
                <div className="p-4 bg-[#8cd139]/10 rounded-lg border border-[#8cd139]/20">
                  <h4 className="font-semibold text-[#2d5016] mb-2">Real-World Success</h4>

        ease: 'back.out(1.4)'
      }, '-=0.6');

      const handleMouseMove = (e: MouseEvent) => {
        const rect = heroRef.current?.getBoundingClientRect();
        if (!rect) return;

        const x = (e.clientX - rect.left) / rect.width - 0.5;
        const y = (e.clientY - rect.top) / rect.height - 0.5;

        // Ciblez les éléments pertinents pour le parallaxe
        gsap.to('.hero-title, .hero-description', {
          x: x * 10,
          y: y * 5,
          duration: 2,
          ease: 'power2.out'
        });

        gsap.to('.hero-image-container', {
          x: x * -10, // Mouvement inversé pour un effet de profondeur
          y: y * -5,
          duration: 2.5,
          ease: 'power2.out'
        });
      };
